/**
 * Lida com o clique no botão Editar.
 * 1. Atualiza o status no Appwrite para 'editing' e define o dono.
 * 2. Carrega a lição e verifica se há rascunhos locais.
 * @param {Object} lesson O objeto completo da lição.
 */
async function handleEditLesson(lesson) {
    console.log("✎ Iniciando processo de edição:", lesson.baseName);
    
    try {
        showLoading('Preparando ambiente de edição...');

        // ============================================================
        // 🆕 PASSO 1: Atualizar Status no Servidor (Bloquear para Edição)
        // ============================================================
        
        // Obtém o usuário atual para assinar a edição
        const user = await account.get();
        const userId = user.$id;

        if (lesson.docId) {
            console.log(`🔒 Atualizando status do post ${lesson.docId} para 'editing'...`);
            
            await databases.updateDocument(
                DATABASE_ID,
                POSTS_COLLECTION_ID,
                lesson.docId,
                {
                    postStatus: 'editing',
                    userEditorId: userId
                }
            );
            console.log("✅ Post atualizado no servidor: Status = editing, Editor = " + userId);
            
            // Atualiza o objeto local lesson para refletir a mudança imediata (opcional, mas bom para consistência)
            lesson.hostId = userId;
            lesson.extraData.postStatus = 'editing';
        } else {
            console.warn("⚠️ Lição sem docId. Pulando atualização de status no servidor.");
        }
        // ============================================================
        // 🏁 FIM DO PASSO 1
        // ============================================================

        // ✅ RESETAR srtData (O resto da função continua igual)
        console.log("🧹 Resetando srtData antes de carregar nova lição");
        srtData.length = 0;
        currentSegment = null;
        undoStack = [];
        redoStack = [];
        
        // ✅ LIMPAR o #subtitleName
        const subtitleNameElement = document.getElementById("subtitleName");
        if (subtitleNameElement) {
            subtitleNameElement.textContent = '';
        }
        
        // ✅ Atualizar appState
        appState.subtitle.name = lesson.srt?.fileName || `${lesson.baseName}.srt`;
        console.log("🔄 appState.subtitle.name:", appState.subtitle.name);
        
        // ✅ Buscar rascunho LOCAL (Dexie) específico para ESTA lição
        const draft = await db.drafts.get(lesson.baseName);
        
        console.log(`🔎 Procurando rascunho local para: "${lesson.baseName}"`);
        
        if (draft) {
            // ... (Lógica de rascunho existente mantida) ...
            const useDraft = confirm(
                `Existe uma versão local não salva para "${lesson.baseName}".\n` +
                `Modificada em: ${new Date(draft.lastModified).toLocaleString()}\n\n` +
                `Deseja restaurar este rascunho local?`
            );
            
            if (useDraft) {
                // ... (Lógica de restaurar rascunho mantida) ...
                if (lesson.video) {
                     // Lógica de carregar vídeo (igual ao código anterior)
                     const cachedLesson = lessonsCache.find(l => l.baseName === lesson.baseName);
                     if (cachedLesson && cachedLesson.video && cachedLesson.video.blobUrl) {
                        appState.video.url = cachedLesson.video.blobUrl;
                        updateAllVideos(cachedLesson.video.blobUrl);
                     } else {
                        const videoUrl = getBackblazePublicUrl(lesson.video.fileId);
                        // ... fetch e blob (igual ao código anterior)
                        // Para brevidade, mantive a lógica exata que você já tinha aqui
                        const videoResponse = await fetch(videoUrl);
                        const videoBlob = await videoResponse.blob();
                        const blobUrl = URL.createObjectURL(videoBlob);
                        if (cachedLesson && cachedLesson.video) cachedLesson.video.blobUrl = blobUrl;
                        appState.video.url = blobUrl;
                        updateAllVideos(blobUrl);
                     }
                }
                
                if (lesson.thumb) {
                    const thumbUrl = getBackblazePublicUrl(lesson.thumb.fileId);
                    updateAllThumbnails(thumbUrl);
                    appState.thumbnail.url = thumbUrl;
                }

                srtData.length = 0;
                srtData.push(...JSON.parse(JSON.stringify(draft.srtData)));
                
                requestAnimationFrame(() => {
                    const videoPlayer = document.getElementById("videoPlayer-editSrt");
                    if (videoPlayer && draft.videoPosition) {
                        videoPlayer.currentTime = draft.videoPosition;
                    }
                });

            } else {
                console.log("🔄 Carregando versão original do servidor");
                await previewLesson(lesson);
                await db.drafts.delete(lesson.baseName);
            }
        } else {
            console.log("📦 Nenhum rascunho local, carregando do servidor...");
            await previewLesson(lesson);
        }

        hideFooter();
        setEditorMode('editor');
        showSection('editSrt');
        renderSegments();

        if (lesson.srt && lesson.srt.fileName) {
            appState.subtitle.name = lesson.srt.fileName;
            const subtitleNameElement = document.getElementById("subtitleName");
            if (subtitleNameElement) {
                subtitleNameElement.textContent = lesson.srt.fileName;
            }
        }

        // Força uma atualização do cache e da UI para refletir que o post saiu da Home e foi para MyPosts
        // (Isso fará o post sumir da Home imediatamente após clicar em editar)
        await refreshLessonsCache();

        hideLoading();

    } catch (error) {
        console.error("Erro ao preparar edição:", error);
        alert('Falha ao preparar edição: ' + error.message);
        hideLoading();
    }
}
