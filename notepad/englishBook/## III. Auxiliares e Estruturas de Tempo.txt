        function renderPage() {
            const root = document.getElementById('content-root');
            let html = `<h1 style="color: #64ffda; border-bottom: 2px solid #722f37; padding-bottom: 10px;">${data.stage_title}</h1>`;

            data.topics.forEach((topic, index) => {
                html += `<section style="margin-bottom: 40px;">
                    <h2 style="color: #f1d3d6; background-color: #1d3354; padding: 10px; border-radius: 4px;">${index + 1}. ${topic.topic_title}</h2>`;

                topic.content.forEach(item => {
                    if (item.type === "subtitle") {
                        html += `<h3 style="color: #ccd6f6; margin-top: 25px;">${item.value}</h3>`;
                    } else if (item.type === "text") {
                        html += `<p>${item.value}</p>`;
                    } else if (item.type === "list") {
                        html += `<ul>${item.value.map(li => `<li>${li}</li>`).join('')}</ul>`;
                    } else if (item.type === "table") {
                        html += `<table style="width: 100%; border-collapse: collapse; margin: 20px 0; background-color: #0a192f;">
                            <thead><tr style="border-bottom: 2px solid #722f37; text-align: left;">
                                ${item.headers.map(h => `<th style="padding: 12px;">${h}</th>`).join('')}
                            </tr></thead>
                            <tbody>
                                ${item.rows.map(row => `<tr style="border-bottom: 1px solid #1d3354;">
                                    ${row.map(cell => `<td style="padding: 10px;">${cell}</td>`).join('')}
                                </tr>`).join('')}
                            </tbody>
                        </table>`;
                    } else if (item.type === "example") {
                        html += `<p style="font-style: italic; background-color: #0a192f; padding: 10px; border-left: 4px solid #64ffda; margin-bottom: 10px;">
                            <strong>EN:</strong> ${item.value.en} | <strong>PT:</strong> ${item.value.pt}
                        </p>`;
                    } else if (item.type === "tip") {
                        html += `<div style="background-color: #722f3733; border: 1px solid #722f37; padding: 15px; border-radius: 4px; margin-top: 15px;">
                            <strong style="color: #f1d3d6;">💡 ${item.title}:</strong> ${item.text}
                        </div>`;
                    }
                });

                // Renderização de Tasks (Quiz, Fill, Connect)
                topic.tasks.forEach(task => {
                    html += `<div style="margin-top: 20px; padding: 15px; border: 1px dashed #64ffda; border-radius: 4px;">`;
                    if (task.tipo === "quiz") {
                        html += `<p><strong>Questão:</strong> ${task.pergunta}</p>`;
                        task.opcoes.forEach(opt => {
                            html += `<label style="display: block; margin-bottom: 5px;">
                                <input type="radio" name="${task.id}"> ${opt.texto} ${opt.correta ? '<span style="color: #64ffda; font-size: 0.8em; margin-left: 10px;">(Correta)</span>' : ''}
                            </label>`;
                        });
                        html += `<p style="font-size: 0.9em; color: #8892b0; margin-top: 10px;"><em>Explicação: ${task.explicacao}</em></p>`;
                    } else if (task.tipo === "fill_the_blank") {
                        html += `<p><strong>Preencha a lacuna:</strong> ${task.pergunta}</p>`;
                        html += `<p style="color: #64ffda;">Resposta correta: ${task.resposta_correta}</p>`;
                    } else if (task.tipo === "connect_pairs") {
                        html += `<p><strong>${task.titulo}:</strong></p><ul style="list-style: none; padding: 0;">`;
                        task.pares.forEach(p => {
                            html += `<li>🔗 ${p.chave} &rarr; ${p.valor}</li>`;
                        });
                        html += `</ul>`;
                    }
                    html += `</div>`;
                });

                html += `</section>`;
            });

            root.innerHTML = html;
        }

        window.onload = renderPage;
